	Rules for this exercise:
	\begin{itemize}
		\item Only integer exponents, no fractional exponents and no radicals.
		\item Negative exponents may appear during calculation, but final answers must contain no negative exponents.
		\item if ans like this \(x^{y}\) , you just write \verb|x^y|
		\item if ans like this \(\frac{x}{y}\) , you just write \verb|\frac{x}{y}|
\end{itemize}

\section{1 Basic Problems}
Simplify each algebraic expression:
\begin{enumerate}
	\item \(x^{5} \cdot x^{3}\)
	\item \(a^{10} ÷ a^{4}\)
	\item \(\left(m^{4}\right)^{3}\)
	\item \((4 y)^{2}\)
	\item \(n^{0} \quad(n \neq 0)\)
	\item \(\frac{k^{3}}{k^{8}}\)
	\item \((3 a b)^{3}\)
	\item \(\frac{p^{2}}{p^{9}}\)
\end{enumerate}

\section{2 Intermediate Mixed Problems}
\begin{enumerate}
	\item \(x^{4} \cdot x ÷ x^{7}\)
	\item \(\left(2 c^{3}\right)^{2} ÷ 4 c^{4}\)
	\item \(\frac{(5 t)^{2}}{t^{6}}\)
	\item \(\left(\frac{x^{4}}{y^{2}}\right)^{3} ÷ x^{15}\)
	\item \(a^{2} \cdot a^{5} ÷ a^{10}\)
	\item \(\frac{\left(2 s^{2}\right)^{4}}{16 s^{9}}\)
	\item \(\left(3 m^{2} n\right)^{2} ÷ m^{7}\)
	\item \(\frac{z^{4} \cdot z}{z^{11}}\)
\end{enumerate}

\section{3 Advanced Comprehensive Problems}
\begin{enumerate}
	\item \(\frac{\left(2 a^{2} b\right)^{3}}{\left(4 a^{4} b^{2}\right)^{2}}\)
	\item \(\left(\frac{3 x^{2}}{y^{3}}\right)^{2} ÷ x^{9}\)
	\item \(\frac{5^{n+4}}{5^{n+7}}\)
	\item \(\left(2 p^{3} q\right)^{2} \cdot p ÷ p^{10}\)
	\item \(\frac{\left(3 m^{4}\right)^{2}}{9 m^{11}}\)
\end{enumerate}

\section{4 Evaluation Problems}
Given \(a^{m}=5\) , \(a^{n}=3\)
\begin{enumerate}
	\item Evaluate \(a^{m+n}\)
	\item Evaluate \(a^{3 m-n}\)
\end{enumerate}
