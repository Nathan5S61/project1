	\section{Level1}
	\subsection*{Part1.Determine whether each equation is an identity. Answer true or false.}
	\begin{enumerate}
		\item You can use the cross method.
		\item Text your answer by true(if yes), false(if not)
	\end{enumerate}
	\begin{enumerate}
		\item $(x+2)^2 = x^2 +4x +4$
		\item $(x-3)^2 = x^2 -9$
		\item $(a+b)(a-b)=a^2-b^2$
		\item $(x+1)(x+2)=x^2+3x$
		\item $10r^2+8=5(2r^2-\dfrac{1}{5})-7$
		\item $x^3-8=(x-2)(x^2+2x+4)$
		\item $(2x-5)^2=4x^2-20x+25$
		\item $x^2-4x = x(x-4)+1$
		\item $(x+y)^3 = x^3+y^3$
		\item $(a+1)(a-6)=a^2-5a-6$
	\end{enumerate}

	\subsection*{Part 5. Word problems}
	\begin{enumerate}
		\item The length of a rectangle is $(3x-4)$ cm and the width is $(2x+5)$ cm. Expand and simplify the expression for its area.
		\item A square has side $(4a-3)$ cm. If each side is decreased by $2$ cm, find the expanded form of the new area.
		\item Given $m+n=15$ and $mn=44$. Use algebraic identity to evaluate $m^2 + n^2$.
		\item Two positive integers differ by $7$, their product is $30$. Let the smaller number be $x$. Set up an equation and expand it.
		\item The area of a rectangle is $6x^2-7x-20$. One side is $(2x-5)$. Find the expression of the other side.
		\item Show that the difference between the squares of two consecutive odd numbers is divisible by $8$. Let the two numbers be $2k-1$ and $2k+1$.
	\end{enumerate}
