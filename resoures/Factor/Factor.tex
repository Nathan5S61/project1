	\section{1 Basic Level (Common factor, difference of squares)}
	Factorize each of the following polynomials:
	\begin{enumerate}
		\item \(3 x+6\)
		\item \(5 a^{2}-10 a\)
		\item \(x^{2}-9\)
		\item \(4 y^{2}-25\)
		\item \(6 m^{2}+9 m\)
	\end{enumerate}
	\section{2 Intermediate Level (Perfect square trinomial, quadratic trinomial)}
	\begin{enumerate}
		\item \(x^{2}+6 x+9\)
		\item \(a^{2}-8 a+16\)
		\item \(x^{2}+5 x+6\)
		\item \(x^{2}-2 x-8\)
		\item \(2 x^{2}+7 x+3\)
	\end{enumerate}
	\section{3 Advanced Level (Grouping method, repeated factorization)}
	\begin{enumerate}
		\item \(a x+a y+b x+b y\)
		\item \(x^{3}-4 x\)
		\item \(x^{2}-y^{2}+x+y\)
		\item \(3 x^{2}-12 x y+12 y^{2}\)
		\item \(x^{4}-16\)
		\item \(x^{3}+3 x^{2}-4 x-12\)
	\end{enumerate}
