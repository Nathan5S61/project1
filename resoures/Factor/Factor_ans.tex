	\section{1 Basic Level Solutions}
	\begin{enumerate}
		\item \(3 x+6=3(x+2)\)
		\item \(5 a^{2}-10 a=5 a(a-2)\)
		\item \(x^{2}-9=(x+3)(x-3)\)
		\item \(4 y^{2}-25=(2 y+5)(2 y-5)\)
		\item \(6 m^{2}+9 m=3 m(2 m+3)\)
	\end{enumerate}
	
	\section{2 Intermediate Level Solutions}
	\begin{enumerate}
		\item \(x^{2}+6 x+9=(x+3)^{2}\)
		\item \(a^{2}-8 a+16=(a-4)^{2}\)
		\item \(x^{2}+5 x+6=(x+2)(x+3)\)
		\item \(x^{2}-2 x-8=(x-4)(x+2)\)
		\item \(2 x^{2}+7 x+3=(2 x+1)(x+3)\)
	\end{enumerate}
	
	\section{3 Advanced Level Solutions}
	\begin{enumerate}
		\item \(a x+a y+b x+b y=(a+b)(x+y)\)
		\item \(x^{3}-4 x=x\left(x^{2}-4\right)=x(x+2)(x-2)\)
		\item \(x^{2}-y^{2}+x+y=(x+y)(x-y)+(x+y)=(x+y)(x-y+1)\)
		\item \(3 x^{2}-12 x y+12 y^{2}=3\left(x^{2}-4 x y+4 y^{2}\right)=3(x-2 y)^{2}\)
		\item \(x^{4}-16=\left(x^{2}+4\right)\left(x^{2}-4\right)=\left(x^{2}+4\right)(x+2)(x-2)\)
		\item \(x^{3}+3 x^{2}-4 x-12=x^{2}(x+3)-4(x+3)=\left(x^{2}-4\right)(x+3)=(x+2)(x-2)(x+3)\)
	\end{enumerate}
